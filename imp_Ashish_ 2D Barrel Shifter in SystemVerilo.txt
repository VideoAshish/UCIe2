# 2D Barrel Shifter in SystemVerilog

A 2D barrel shifter operates on a 2D array of data, performing shifts in both horizontal (within words) and vertical (across words) directions. Below are multiple implementations with increasing complexity.

## 1. Combinatorial 2D Barrel Shifter (Simple)

```systemverilog
//==============================================================================
// Simple 2D Barrel Shifter - Combinatorial
//==============================================================================
// Performs horizontal and vertical shifts using built-in operators
//==============================================================================
module barrel_shifter_2d_simple #(
    parameter int NUM_ROWS = 4,                    // Number of words
    parameter int NUM_COLS = 32,                   // Bits per word
    parameter int SHAMT_H_WIDTH = $clog2(NUM_COLS), // Horizontal shift width
    parameter int SHAMT_V_WIDTH = $clog2(NUM_ROWS)  // Vertical shift width
)(
    input  logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_in,
    input  logic [SHAMT_H_WIDTH-1:0]          shamt_h,
    input  logic [SHAMT_V_WIDTH-1:0]          shamt_v,
    input  logic                              left_h,     // 0: right, 1: left
    input  logic                              up_v,       // 0: down, 1: up  
    input  logic                              arithmetic, // 0: logical, 1: arithmetic
    
    output logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_out
);

    // Internal signal after horizontal shift
    logic [NUM_ROWS-1:0][NUM_COLS-1:0] h_shifted;
    
    //==========================================================================
    // Horizontal Shift Stage - Shift each row independently
    //==========================================================================
    always_comb begin
        for (int row = 0; row < NUM_ROWS; row++) begin
            if (left_h) begin
                // Left shift (always logical)
                h_shifted[row] = data_in[row] << shamt_h;
            end else begin
                // Right shift
                if (arithmetic) begin
                    h_shifted[row] = data_in[row] >>> shamt_h;
                end else begin
                    h_shifted[row] = data_in[row] >> shamt_h;
                end
            end
        end
    end
    
    //==========================================================================
    // Vertical Shift Stage - Shift rows up or down
    //==========================================================================
    always_comb begin
        for (int row = 0; row < NUM_ROWS; row++) begin
            int src_row;
            
            if (up_v) begin
                // Shift up: row[i] gets data from row[i + shamt_v]
                src_row = row + shamt_v;
                if (src_row < NUM_ROWS) begin
                    data_out[row] = h_shifted[src_row];
                end else if (arithmetic) begin
                    // Arithmetic: sign extend from top row
                    data_out[row] = {NUM_COLS{h_shifted[NUM_ROWS-1][NUM_COLS-1]}};
                end else begin
                    data_out[row] = '0;
                end
            end else begin
                // Shift down: row[i] gets data from row[i - shamt_v]
                src_row = row - shamt_v;
                if (src_row >= 0) begin
                    data_out[row] = h_shifted[src_row];
                end else if (arithmetic) begin
                    // Arithmetic: sign extend from bottom row
                    data_out[row] = {NUM_COLS{h_shifted[0][NUM_COLS-1]}};
                end else begin
                    data_out[row] = '0;
                end
            end
        end
    end

endmodule
```

## 2. Logarithmic 2D Barrel Shifter (Pipeline)

```systemverilog
//==============================================================================
// Logarithmic 2D Barrel Shifter - Pipelined
//==============================================================================
// Uses logarithmic shift stages for optimal timing
// Each stage shifts by power-of-2 amounts: 1, 2, 4, 8, ...
//==============================================================================
module barrel_shifter_2d_log #(
    parameter int NUM_ROWS = 4,
    parameter int NUM_COLS = 32,
    parameter int SHAMT_H_WIDTH = $clog2(NUM_COLS),
    parameter int SHAMT_V_WIDTH = $clog2(NUM_ROWS)
)(
    input  logic                              clk,
    input  logic                              rst_n,
    input  logic                              valid_in,
    input  logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_in,
    input  logic [SHAMT_H_WIDTH-1:0]          shamt_h,
    input  logic [SHAMT_V_WIDTH-1:0]          shamt_v,
    input  logic                              left_h,
    input  logic                              up_v,
    input  logic                              arithmetic,
    
    output logic                              valid_out,
    output logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_out
);

    localparam int H_STAGES = SHAMT_H_WIDTH;
    localparam int V_STAGES = SHAMT_V_WIDTH;
    localparam int TOTAL_STAGES = H_STAGES + V_STAGES + 2; // +2 for input/output regs
    
    // Pipeline registers for data
    logic [NUM_ROWS-1:0][NUM_COLS-1:0] pipe_data [0:TOTAL_STAGES-1];
    
    // Pipeline registers for control signals
    logic [SHAMT_H_WIDTH-1:0] pipe_shamt_h [0:H_STAGES];
    logic [SHAMT_V_WIDTH-1:0] pipe_shamt_v [0:V_STAGES];
    logic pipe_left_h [0:H_STAGES];
    logic pipe_up_v [0:V_STAGES];
    logic pipe_arith [0:TOTAL_STAGES-1];
    logic pipe_valid [0:TOTAL_STAGES-1];
    
    //==========================================================================
    // Input Pipeline Stage
    //==========================================================================
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            pipe_data[0] <= '0;
            pipe_shamt_h[0] <= '0;
            pipe_left_h[0] <= '0;
            pipe_arith[0] <= '0;
            pipe_valid[0] <= 1'b0;
        end else begin
            pipe_data[0] <= data_in;
            pipe_shamt_h[0] <= shamt_h;
            pipe_left_h[0] <= left_h;
            pipe_arith[0] <= arithmetic;
            pipe_valid[0] <= valid_in;
        end
    end
    
    //==========================================================================
    // Horizontal Shift Stages (Logarithmic)
    //==========================================================================
    generate
        for (genvar s = 0; s < H_STAGES; s++) begin : h_stage_gen
            localparam int SHIFT_AMT = (1 << s);
            
            always_ff @(posedge clk or negedge rst_n) begin
                if (!rst_n) begin
                    pipe_data[s+1] <= '0;
                    pipe_shamt_h[s+1] <= '0;
                    pipe_left_h[s+1] <= '0;
                    pipe_arith[s+1] <= '0;
                    pipe_valid[s+1] <= 1'b0;
                end else begin
                    // Propagate control signals
                    pipe_shamt_h[s+1] <= pipe_shamt_h[s];
                    pipe_left_h[s+1] <= pipe_left_h[s];
                    pipe_arith[s+1] <= pipe_arith[s];
                    pipe_valid[s+1] <= pipe_valid[s];
                    
                    for (int row = 0; row < NUM_ROWS; row++) begin
                        if (pipe_shamt_h[s][s]) begin
                            if (pipe_left_h[s]) begin
                                // Left shift by 2^s positions
                                if (SHIFT_AMT >= NUM_COLS) begin
                                    pipe_data[s+1][row] <= '0;
                                end else begin
                                    pipe_data[s+1][row][NUM_COLS-1:SHIFT_AMT] <= 
                                        pipe_data[s][row][NUM_COLS-1-SHIFT_AMT:0];
                                    pipe_data[s+1][row][SHIFT_AMT-1:0] <= '0;
                                end
                            end else begin
                                // Right shift by 2^s positions
                                if (SHIFT_AMT >= NUM_COLS) begin
                                    if (pipe_arith[s]) begin
                                        pipe_data[s+1][row] <= {NUM_COLS{pipe_data[s][row][NUM_COLS-1]}};
                                    end else begin
                                        pipe_data[s+1][row] <= '0;
                                    end
                                end else begin
                                    pipe_data[s+1][row][NUM_COLS-1-SHIFT_AMT:0] <= 
                                        pipe_data[s][row][NUM_COLS-1:SHIFT_AMT];
                                    if (pipe_arith[s]) begin
                                        pipe_data[s+1][row][NUM_COLS-1:NUM_COLS-SHIFT_AMT] <= 
                                            {SHIFT_AMT{pipe_data[s][row][NUM_COLS-1]}};
                                    end else begin
                                        pipe_data[s+1][row][NUM_COLS-1:NUM_COLS-SHIFT_AMT] <= '0;
                                    end
                                end
                            end
                        end else begin
                            // Pass through
                            pipe_data[s+1][row] <= pipe_data[s][row];
                        end
                    end
                end
            end
        end
    endgenerate
    
    //==========================================================================
    // Vertical Shift Control Capture
    //==========================================================================
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            pipe_shamt_v[0] <= '0;
            pipe_up_v[0] <= '0;
        end else begin
            pipe_shamt_v[0] <= shamt_v;
            pipe_up_v[0] <= up_v;
        end
    end
    
    // Propagate vertical control through horizontal stages
    generate
        for (genvar s = 0; s < H_STAGES; s++) begin : v_ctrl_prop
            always_ff @(posedge clk or negedge rst_n) begin
                if (!rst_n) begin
                    pipe_shamt_v[s+1] <= '0;
                    pipe_up_v[s+1] <= '0;
                end else begin
                    pipe_shamt_v[s+1] <= pipe_shamt_v[s];
                    pipe_up_v[s+1] <= pipe_up_v[s];
                end
            end
        end
    endgenerate
    
    //==========================================================================
    // Vertical Shift Stages (Logarithmic)
    //==========================================================================
    generate
        for (genvar s = 0; s < V_STAGES; s++) begin : v_stage_gen
            localparam int SHIFT_AMT = (1 << s);
            localparam int STAGE_IDX = H_STAGES + 1 + s;
            
            always_ff @(posedge clk or negedge rst_n) begin
                if (!rst_n) begin
                    pipe_data[STAGE_IDX] <= '0;
                    pipe_shamt_v[s+1] <= '0;
                    pipe_up_v[s+1] <= '0;
                    pipe_arith[STAGE_IDX] <= '0;
                    pipe_valid[STAGE_IDX] <= 1'b0;
                end else begin
                    pipe_shamt_v[s+1] <= pipe_shamt_v[s];
                    pipe_up_v[s+1] <= pipe_up_v[s];
                    pipe_arith[STAGE_IDX] <= pipe_arith[STAGE_IDX-1];
                    pipe_valid[STAGE_IDX] <= pipe_valid[STAGE_IDX-1];
                    
                    for (int row = 0; row < NUM_ROWS; row++) begin
                        if (pipe_shamt_v[s][s]) begin
                            int src_row;
                            
                            if (pipe_up_v[s]) begin
                                // Shift up: row[i] = row[i + 2^s]
                                src_row = row + SHIFT_AMT;
                                if (src_row < NUM_ROWS) begin
                                    pipe_data[STAGE_IDX][row] <= pipe_data[STAGE_IDX-1][src_row];
                                end else if (pipe_arith[STAGE_IDX]) begin
                                    pipe_data[STAGE_IDX][row] <= 
                                        {NUM_COLS{pipe_data[STAGE_IDX-1][NUM_ROWS-1][NUM_COLS-1]}};
                                end else begin
                                    pipe_data[STAGE_IDX][row] <= '0;
                                end
                            end else begin
                                // Shift down: row[i] = row[i - 2^s]
                                src_row = row - SHIFT_AMT;
                                if (src_row >= 0) begin
                                    pipe_data[STAGE_IDX][row] <= pipe_data[STAGE_IDX-1][src_row];
                                end else if (pipe_arith[STAGE_IDX]) begin
                                    pipe_data[STAGE_IDX][row] <= 
                                        {NUM_COLS{pipe_data[STAGE_IDX-1][0][NUM_COLS-1]}};
                                end else begin
                                    pipe_data[STAGE_IDX][row] <= '0;
                                end
                            end
                        end else begin
                            pipe_data[STAGE_IDX][row] <= pipe_data[STAGE_IDX-1][row];
                        end
                    end
                end
            end
        end
    endgenerate
    
    //==========================================================================
    // Output Pipeline Stage
    //==========================================================================
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out <= '0;
            valid_out <= 1'b0;
        end else begin
            data_out <= pipe_data[TOTAL_STAGES-1];
            valid_out <= pipe_valid[TOTAL_STAGES-1];
        end
    end

endmodule
```

## 3. 2D Barrel Shifter with Wrap-Around Support

```systemverilog
//==============================================================================
// 2D Barrel Shifter with Wrap-Around (Toroidal Shift)
//==============================================================================
// Supports circular shifts in both dimensions
//==============================================================================
module barrel_shifter_2d_wrap #(
    parameter int NUM_ROWS = 4,
    parameter int NUM_COLS = 32,
    parameter int SHAMT_H_WIDTH = $clog2(NUM_COLS),
    parameter int SHAMT_V_WIDTH = $clog2(NUM_ROWS)
)(
    input  logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_in,
    input  logic [SHAMT_H_WIDTH-1:0]          shamt_h,
    input  logic [SHAMT_V_WIDTH-1:0]          shamt_v,
    input  logic                              left_h,
    input  logic                              up_v,
    input  logic                              wrap_h,     // Enable horizontal wrap
    input  logic                              wrap_v,     // Enable vertical wrap
    input  logic                              arithmetic,
    
    output logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_out
);

    // Flatten 2D array for easier circular manipulation
    logic [NUM_ROWS*NUM_COLS-1:0] flat_in, flat_out;
    logic [SHAMT_H_WIDTH+SHAMT_V_WIDTH-1:0] total_shift;
    
    //==========================================================================
    // Flatten 2D input to 1D
    // Row 0 is MSBs, Row NUM_ROWS-1 is LSBs
    //==========================================================================
    always_comb begin
        for (int row = 0; row < NUM_ROWS; row++) begin
            flat_in[row*NUM_COLS +: NUM_COLS] = data_in[row];
        end
    end
    
    //==========================================================================
    // Calculate combined shift amount
    // For wrap-around: total_shift = vertical_shift * NUM_COLS + horizontal_shift
    //==========================================================================
    always_comb begin
        if (up_v) begin
            // Shift up means lower rows move to higher positions
            // In flat representation, this is a left shift
            if (left_h) begin
                total_shift = shamt_v * NUM_COLS + shamt_h;
            end else begin
                total_shift = shamt_v * NUM_COLS - shamt_h;
            end
        end else begin
            // Shift down means higher rows move to lower positions
            // In flat representation, this is a right shift
            if (left_h) begin
                total_shift = -(int'(shamt_v) * NUM_COLS) + shamt_h;
            end else begin
                total_shift = -(int'(shamt_v) * NUM_COLS) - shamt_h;
            end
        end
    end
    
    //==========================================================================
    // Perform the shift with wrap-around support
    //==========================================================================
    localparam int TOTAL_BITS = NUM_ROWS * NUM_COLS;
    localparam int TOTAL_SHIFT_WIDTH = $clog2(TOTAL_BITS) + 1; // +1 for sign
    
    always_comb begin
        if (wrap_h && wrap_v) begin
            // Full 2D wrap-around (circular shift)
            logic [TOTAL_BITS-1:0] shifted;
            shifted = (flat_in <<< total_shift[TOTAL_SHIFT_WIDTH-1:0]);
            flat_out = shifted;
        end else if (wrap_v) begin
            // Only vertical wrap, horizontal zero/sign fill
            logic [NUM_ROWS-1:0][NUM_COLS-1:0] v_shifted;
            
            for (int row = 0; row < NUM_ROWS; row++) begin
                int src_row;
                if (up_v) begin
                    src_row = (row + shamt_v) % NUM_ROWS;
                end else begin
                    src_row = (row - shamt_v + NUM_ROWS) % NUM_ROWS;
                end
                v_shifted[row] = data_in[src_row];
            end
            
            // Apply horizontal shift
            for (int row = 0; row < NUM_ROWS; row++) begin
                if (left_h) begin
                    flat_out[row*NUM_COLS +: NUM_COLS] = v_shifted[row] << shamt_h;
                end else begin
                    if (arithmetic) begin
                        flat_out[row*NUM_COLS +: NUM_COLS] = v_shifted[row] >>> shamt_h;
                    end else begin
                        flat_out[row*NUM_COLS +: NUM_COLS] = v_shifted[row] >> shamt_h;
                    end
                end
            end
        end else if (wrap_h) begin
            // Only horizontal wrap, vertical zero/sign fill
            logic [NUM_ROWS-1:0][NUM_COLS-1:0] h_shifted;
            
            for (int row = 0; row < NUM_ROWS; row++) begin
                h_shifted[row] = (data_in[row] <<< shamt_h);
            end
            
            // Apply vertical shift
            for (int row = 0; row < NUM_ROWS; row++) begin
                int src_row;
                if (up_v) begin
                    src_row = row + shamt_v;
                end else begin
                    src_row = row - shamt_v;
                end
                
                if (src_row >= 0 && src_row < NUM_ROWS) begin
                    flat_out[row*NUM_COLS +: NUM_COLS] = h_shifted[src_row];
                end else if (arithmetic) begin
                    flat_out[row*NUM_COLS +: NUM_COLS] = 
                        {NUM_COLS{h_shifted[up_v ? NUM_ROWS-1 : 0][NUM_COLS-1]}};
                end else begin
                    flat_out[row*NUM_COLS +: NUM_COLS] = '0;
                end
            end
        end else begin
            // No wrap - standard shift with zero/sign fill
            logic [NUM_ROWS-1:0][NUM_COLS-1:0] h_shifted;
            logic [NUM_ROWS-1:0][NUM_COLS-1:0] v_shifted;
            
            for (int row = 0; row < NUM_ROWS; row++) begin
                if (left_h) begin
                    h_shifted[row] = data_in[row] << shamt_h;
                end else begin
                    if (arithmetic) begin
                        h_shifted[row] = data_in[row] >>> shamt_h;
                    end else begin
                        h_shifted[row] = data_in[row] >> shamt_h;
                    end
                end
            end
            
            for (int row = 0; row < NUM_ROWS; row++) begin
                int src_row;
                if (up_v) begin
                    src_row = row + shamt_v;
                end else begin
                    src_row = row - shamt_v;
                end
                
                if (src_row >= 0 && src_row < NUM_ROWS) begin
                    v_shifted[row] = h_shifted[src_row];
                end else if (arithmetic) begin
                    v_shifted[row] = {NUM_COLS{h_shifted[up_v ? NUM_ROWS-1 : 0][NUM_COLS-1]}};
                end else begin
                    v_shifted[row] = '0;
                end
                flat_out[row*NUM_COLS +: NUM_COLS] = v_shifted[row];
            end
        end
    end
    
    //==========================================================================
    // Unflatten 1D output to 2D
    //==========================================================================
    always_comb begin
        for (int row = 0; row < NUM_ROWS; row++) begin
            data_out[row] = flat_out[row*NUM_COLS +: NUM_COLS];
        end
    end

endmodule
```

## 4. Testbench

```systemverilog
//==============================================================================
// Testbench for 2D Barrel Shifter
//==============================================================================
module tb_barrel_shifter_2d;
    
    // Parameters
    localparam int NUM_ROWS = 4;
    localparam int NUM_COLS = 8;
    localparam int SHAMT_H_WIDTH = $clog2(NUM_COLS);
    localparam int SHAMT_V_WIDTH = $clog2(NUM_ROWS);
    
    // Signals - Simple shifter
    logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_in_s;
    logic [SHAMT_H_WIDTH-1:0]          shamt_h_s;
    logic [SHAMT_V_WIDTH-1:0]          shamt_v_s;
    logic                              left_h_s, up_v_s, arithmetic_s;
    logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_out_s;
    
    // Signals - Log shifter
    logic                              clk, rst_n, valid_in;
    logic                              valid_out;
    logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_out_log;
    
    // Signals - Wrap shifter
    logic [NUM_ROWS-1:0][NUM_COLS-1:0] data_out_wrap;
    logic                              wrap_h, wrap_v;
    
    // Instantiate DUTs
    barrel_shifter_2d_simple #(
        .NUM_ROWS(NUM_ROWS),
        .NUM_COLS(NUM_COLS)
    ) uut_simple (
        .data_in(data_in_s),
        .shamt_h(shamt_h_s),
        .shamt_v(shamt_v_s),
        .left_h(left_h_s),
        .up_v(up_v_s),
        .arithmetic(arithmetic_s),
        .data_out(data_out_s)
    );
    
    barrel_shifter_2d_log #(
        .NUM_ROWS(NUM_ROWS),
        .NUM_COLS(NUM_COLS)
    ) uut_log (
        .clk(clk),
        .rst_n(rst_n),
        .valid_in(valid_in),
        .data_in(data_in_s),
        .shamt_h(shamt_h_s),
        .shamt_v(shamt_v_s),
        .left_h(left_h_s),
        .up_v(up_v_s),
        .arithmetic(arithmetic_s),
        .valid_out(valid_out),
        .data_out(data_out_log)
    );
    
    barrel_shifter_2d_wrap #(
        .NUM_ROWS(NUM_ROWS),
        .NUM_COLS(NUM_COLS)
    ) uut_wrap (
        .data_in(data_in_s),
        .shamt_h(shamt_h_s),
        .shamt_v(shamt_v_s),
        .left_h(left_h_s),
        .up_v(up_v_s),
        .wrap_h(wrap_h),
        .wrap_v(wrap_v),
        .arithmetic(arithmetic_s),
        .data_out(data_out_wrap)
    );
    
    // Clock generation
    initial begin
        clk = 0;
        forever #5 clk = ~clk;
    end
    
    // Test procedure
    initial begin
        // Initialize
        rst_n = 0;
        valid_in = 0;
        wrap_h = 0;
        wrap_v = 0;
        arithmetic_s = 0;
        left_h_s = 0;
        up_v_s = 0;
        shamt_h_s = 0;
        shamt_v_s = 0;
        data_in_s = '0;
        
        #20 rst_n = 1;
        #10;
        
        //----------------------------------------------------------------------
        // Test 1: Initialize with pattern data
        //----------------------------------------------------------------------
        $display("\n=== Test 1: Basic Pattern ===");
        for (int i = 0; i < NUM_ROWS; i++) begin
            data_in_s[i] = i * 16 + 8'hAA;  // Each row has unique value
        end
        display_data("Input:", data_in_s);
        
        //----------------------------------------------------------------------
        // Test 2: No shift
        //----------------------------------------------------------------------
        $display("\n=== Test 2: No Shift ===");
        shamt_h_s = 0;
        shamt_v_s = 0;
        left_h_s = 0;
        up_v_s = 0;
        valid_in = 1;
        #10;
        valid_in = 0;
        display_data("Output:", data_out_s);
        
        //----------------------------------------------------------------------
        // Test 3: Horizontal right shift by 2
        //----------------------------------------------------------------------
        $display("\n=== Test 3: Horizontal Right Shift by 2 ===");
        shamt_h_s = 2;
        shamt_v_s = 0;
        left_h_s = 0;
        valid_in = 1;
        #10;
        valid_in = 0;
        display_data("Output:", data_out_s);
        
        //----------------------------------------------------------------------
        // Test 4: Horizontal left shift by 3
        //----------------------------------------------------------------------
        $display("\n=== Test 4: Horizontal Left Shift by 3 ===");
        shamt_h_s = 3;
        shamt_v_s = 0;
        left_h_s = 1;
        valid_in = 1;
        #10;
        valid_in = 0;
        display_data("Output:", data_out_s);
        
        //----------------------------------------------------------------------
        // Test 5: Vertical shift down by 1
        //----------------------------------------------------------------------

        shamt_v_s = 1;
        up_v_s = 0;
        valid_in = 1;
        #10;
        valid_in = 0;
        display_data("Output:", data_out_s);
        
        //----------------------------------------------------------------------
        // Test 6: Vertical shift up by 2
        //----------------------------------------------------------------------
        $display("\n=== Test 6: Vertical Shift Up by 2 ===");
        shamt_h_s = 0;
        shamt_v_s = 2;
        up_v_s = 1;
        valid_in = 1;
        #10;
        valid_in = 0;
        display_data("Output:", data_out_s);
        
        //----------------------------------------------------------------------
        // Test 7: Combined horizontal and vertical shift
        //----------------------------------------------------------------------
        $display("\n=== Test 7: Combined H-Right by 2, V-Down by 1 ===");
        shamt_h_s = 2;
        shamt_v_s = 1;
```
